109: def robust_source_scale(cube, train_coords, quantile=0.95):
110:     if len(train_coords) == 0:

===== core/data_hfcst.py:133 _assign_source_blocks =====
133: def _assign_source_blocks(blocks, gt, num_classes, ratios, rng):
134:     # Greedy global block assignment uses class histograms only to balance splits;
135:     # a block is never split by class, so cross-split patch support cannot overlap.
136:     split_names = list(ratios)
137:     hist = []
138:     for _, r0, r1, c0, c1 in blocks:
139:         labels = gt[r0:r1, c0:c1]
140:         h = np.bincount(labels[(labels > 0) & (labels <= num_classes)], minlength=num_classes + 1)[1:]
141:         hist.append(h.astype(float))
142:     hist = np.asarray(hist) if hist else np.zeros((0, num_classes))
143:     nblocks = len(blocks)
144:     desired_blocks = {s: ratios[s] * nblocks for s in split_names}
145:     desired_class = {s: hist.sum(axis=0) * ratios[s] for s in split_names}
146:     assigned_blocks = {s: 0 for s in split_names}
147:     assigned_class = {s: np.zeros(num_classes, dtype=float) for s in split_names}
148:     jitter = rng.random(nblocks)
149:     order = sorted(range(nblocks), key=lambda i: (hist[i].sum(), jitter[i]), reverse=True)
150:     assignment = {}
151:     for i in order:
152:         eligible = [s for s in split_names if assigned_blocks[s] < math.ceil(desired_blocks[s])]
153:         if not eligible:
154:             eligible = split_names
155:         scores = []
156:         for s in eligible:
157:             deficit = np.maximum(desired_class[s] - assigned_class[s], 0.0)
158:             class_score = float((np.minimum(hist[i], deficit) / np.maximum(desired_class[s], 1.0)).sum())
159:             block_score = (desired_blocks[s] - assigned_blocks[s]) / max(desired_blocks[s], 1.0)
160:             scores.append(class_score + 0.35 * block_score + float(rng.random()) * 1e-5)
161:         chosen = eligible[int(np.argmax(scores))]
162:         assignment[blocks[i][0]] = chosen
163:         assigned_blocks[chosen] += 1
164:         assigned_class[chosen] += hist[i]
165:     return assignment, {"block_counts": assigned_blocks,
166:                         "class_pixel_counts_by_block_split": {s: assigned_class[s].astype(int).tolist() for s in split_names}}

182: def global_spatial_split(gt, valid_mask, num_classes, patch_size=11, grid_size=8,
183:                          seed=1237, train_cap=150, constraint_cap=12,
184:                          calibration_cap=80, validation_cap=150):
185:     """Source split into four globally disjoint block sets; patch supports stay inside blocks."""
186:     H, W = gt.shape
187:     if valid_mask.shape != gt.shape:
188:         raise ValueError("valid_mask and GT shapes differ")
189:     rng = np.random.default_rng(seed)
190:     blocks = _block_bounds(H, W, patch_size, grid_size)
191:     ratios = {"train": 0.55, "constraint": 0.15, "calibration": 0.15, "validation": 0.15}
192:     assignment, block_audit = _assign_source_blocks(blocks, gt, num_classes, ratios, rng)
193:     radius = patch_size // 2
194:     coords_by_split = {s: [] for s in ratios}
195:     block_ids_by_split = {s: [] for s in ratios}
196:     for bid, r0, r1, c0, c1 in blocks:
197:         split = assignment[bid]
198:         rr0, rr1 = r0 + radius, r1 - radius
199:         cc0, cc1 = c0 + radius, c1 - radius
200:         if rr1 <= rr0 or cc1 <= cc0:
201:             continue
202:         rr, cc = np.mgrid[rr0:rr1, cc0:cc1]
203:         candidate = np.stack([rr.reshape(-1), cc.reshape(-1)], axis=1)
204:         y = gt[candidate[:, 0], candidate[:, 1]]
205:         ok = valid_mask[candidate[:, 0], candidate[:, 1]] & (y > 0) & (y <= num_classes)
206:         candidate = candidate[ok]
207:         if len(candidate):
208:             coords_by_split[split].append(candidate)
209:             block_ids_by_split[split].append(np.full(len(candidate), bid, dtype=np.int64))
210:     coords = {}
211:     block_ids = {}
212:     caps = {"train": train_cap, "constraint": constraint_cap,
213:             "calibration": calibration_cap, "validation": validation_cap}
214:     for s, parts in coords_by_split.items():
215:         allc = np.concatenate(parts, axis=0) if parts else np.empty((0, 2), dtype=int)
216:         allb = np.concatenate(block_ids_by_split[s]) if block_ids_by_split[s] else np.empty(0, dtype=np.int64)
217:         if s == "train" and caps[s] and caps[s] > 0:
218:             # Cap per class while preserving block provenance.
219:             selected = []
220:             yy = gt[allc[:, 0], allc[:, 1]]
221:             for c in np.unique(yy):
222:                 ix = np.flatnonzero(yy == c)
223:                 if len(ix) > caps[s]: ix = rng.choice(ix, int(caps[s]), replace=False)
224:                 selected.extend(ix.tolist())
225:             selected = np.asarray(sorted(selected), dtype=int)
226:             allc, allb = allc[selected], allb[selected]
227:         elif s == "constraint" and caps[s] and caps[s] > 0:
228:             selected = []
229:             yy = gt[allc[:, 0], allc[:, 1]]
230:             for c in np.unique(yy):
231:                 ix = np.flatnonzero(yy == c)
232:                 if len(ix) > caps[s]: ix = rng.choice(ix, int(caps[s]), replace=False)
233:                 selected.extend(ix.tolist())
234:             selected = np.asarray(sorted(selected), dtype=int)
235:             allc, allb = allc[selected], allb[selected]
236:         elif s == "calibration" and caps[s] and caps[s] > 0:
237:             selected = []
238:             yy = gt[allc[:, 0], allc[:, 1]]
239:             for c in np.unique(yy):
240:                 ix = np.flatnonzero(yy == c)
241:                 if len(ix) > caps[s]: ix = rng.choice(ix, int(caps[s]), replace=False)
242:                 selected.extend(ix.tolist())
243:             selected = np.asarray(sorted(selected), dtype=int)
244:             allc, allb = allc[selected], allb[selected]
245:         elif s == "validation" and caps[s] and caps[s] > 0:
246:             selected = []
247:             yy = gt[allc[:, 0], allc[:, 1]]
248:             for c in np.unique(yy):
249:                 ix = np.flatnonzero(yy == c)
250:                 if len(ix) > caps[s]: ix = rng.choice(ix, int(caps[s]), replace=False)
251:                 selected.extend(ix.tolist())
252:             selected = np.asarray(sorted(selected), dtype=int)
253:             allc, allb = allc[selected], allb[selected]
254:         coords[s], block_ids[s] = allc, allb
255:     class_counts = {}
256:     for s, xy in coords.items():
257:         y = gt[xy[:, 0], xy[:, 1]] if len(xy) else np.empty(0, dtype=int)
258:         class_counts[s] = {str(c): int((y == c).sum()) for c in range(1, num_classes + 1)}
259:     audit = {"block_count": len(blocks), "block_assignment": {str(k): v for k, v in assignment.items()},
260:              "block_audit": block_audit, "class_counts": class_counts,
261:              "patch_size": patch_size, "patch_support_disjoint_across_splits": True,
262:              "note": "Each center is at least patch_radius from its assigned block boundary."}
263:     return coords, block_ids, audit
